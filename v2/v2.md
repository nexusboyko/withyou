import { useState } from "react";
import { motion, useMotionValue, useTransform } from "framer-motion";

// ---------- Color palette ----------
const PALETTE = {
  b1: "#2F5D50", // forest
  b2: "#3A6F8F", // sea
  b3: "#E3C77A", // coast
  b4: "#B25C4A", // clay
  b5: "#8E8E8E", // rock
} as const;

// ---------- Shared sizing ----------
const BALL = 56;
const ORBIT_R = 88; // Version D radius

// ---------- Version D rest layout (original design) ----------
const restLayout = {
  b1: { x: 140, y: 160, s: 1 },
  b2: { x: 200, y: 160, s: 1 },
  b3: { x: 140, y: 220, s: 1 },
  b4: { x: 200, y: 220, s: 1 },
  b5: { x: 170, y: 100, s: 1 },
} as const;

type BallId = keyof typeof restLayout;
const BALLS: BallId[] = ["b1", "b2", "b3", "b4", "b5"];

function Ball({ id, x, y, s, onClick }: { id: BallId; x: number; y: number; s: number; onClick?: () => void }) {
  return (
    <motion.div
      className="absolute rounded-full cursor-pointer"
      style={{ width: BALL, height: BALL, background: PALETTE[id] }}
      animate={{ x, y, scale: s }}
      transition={{ type: "spring", stiffness: 240, damping: 20 }}
      onClick={onClick}
    />
  );
}

// ============================================================
// Version D — Shortest-path encircle (legacy relaxAngles version)
// ============================================================
const MIN_ANGLE = (2 * Math.PI) / 5 * 0.8; // soft minimum separation

function relaxAngles(angles: number[], minSep: number) {
  const a = [...angles];
  const n = a.length;
  // normalize to [0, 2pi)
  for (let i = 0; i < n; i++) a[i] = (a[i] + Math.PI * 2) % (Math.PI * 2);
  // sort with original indices
  const idx = a.map((v, i) => ({ v, i })).sort((x, y) => x.v - y.v);
  // relax in a few passes
  for (let pass = 0; pass < 4; pass++) {
    for (let i = 0; i < n; i++) {
      const j = (i + 1) % n;
      const cur = idx[i].v;
      let next = idx[j].v;
      if (j === 0) next += Math.PI * 2;
      const d = next - cur;
      if (d < minSep) {
        const push = (minSep - d) / 2;
        idx[i].v -= push;
        idx[j].v += push;
      }
    }
  }
  // restore order
  const out = new Array(n).fill(0);
  idx.forEach(({ v, i }) => (out[i] = v));
  return out;
}

function VersionD() {
  const [active, setActive] = useState<BallId | "rest">("rest");

  const layout = (() => {
    if (active === "rest") return restLayout;
    const center = restLayout[active];
    const others = BALLS.filter((b) => b !== active);

    // initial target angles = direction from center to each ball's home (shortest path)
    const rawAngles = others.map((id) => {
      const h = restLayout[id];
      return Math.atan2(h.y - center.y, h.x - center.x);
    });

    const fixedAngles = relaxAngles(rawAngles, MIN_ANGLE);

    const out: any = { ...restLayout };
    out[active] = { ...center, s: 1.4 };

    others.forEach((id, i) => {
      const a = fixedAngles[i];
      out[id] = {
        x: center.x + Math.cos(a) * ORBIT_R,
        y: center.y + Math.sin(a) * ORBIT_R,
        s: 1,
      };
    });

    return out;
  })();

  return (
    <div className="relative" style={{ width: 360, height: 360 }}>
      {BALLS.map((id) => {
        const p = layout[id];
        return <Ball key={id} id={id} x={p.x} y={p.y} s={p.s} onClick={() => setActive((v) => (v === id ? "rest" : id))} />;
      })}
    </div>
  );
}

// ============================================================
// Version F — Bearing / Eye rotation demo (free rolling)
// ============================================================
function BearingDemo() {
  const center = restLayout.b5;
  const angle = useMotionValue(0);
  const angularVelocity = useMotionValue(0);

  const cx = center.x + BALL * 0.2;
  const cy = center.y + BALL * 0.2;

  const x = useTransform(angle, (a) => cx + Math.cos(a) * ORBIT_R);
  const y = useTransform(angle, (a) => cy + Math.sin(a) * ORBIT_R);

  let lastX = 0;
  let lastY = 0;
  let lastTime = 0;

  return (
    <div className="relative" style={{ width: 360, height: 360 }}>
      <motion.div
        className="absolute rounded-full"
        style={{
          width: BALL * 1.4,
          height: BALL * 1.4,
          background: PALETTE.b5,
          x: center.x - BALL * 0.2,
          y: center.y - BALL * 0.2,
        }}
        onClick={() => angularVelocity.set(6)}
        whileTap={{ scale: 1.05 }}
      />

      <motion.div
        className="absolute rounded-full cursor-grab"
        style={{ width: BALL * 0.8, height: BALL * 0.8, background: PALETTE.b2, x, y }}
        onPointerDown={(e) => {
          (e.target as HTMLElement).setPointerCapture(e.pointerId);
          lastX = e.clientX;
          lastY = e.clientY;
          lastTime = performance.now();
          angularVelocity.set(0);
        }}
        onPointerMove={(e) => {
          if ((e.target as HTMLElement).hasPointerCapture(e.pointerId)) {
            const now = performance.now();
            const dx = e.clientX - lastX;
            const dy = e.clientY - lastY;
            const deltaAngle = (dx * -Math.sin(angle.get()) + dy * Math.cos(angle.get())) / ORBIT_R;
            angle.set(angle.get() + deltaAngle);
            angularVelocity.set(deltaAngle / Math.max(16, now - lastTime));
            lastX = e.clientX;
            lastY = e.clientY;
            lastTime = now;
          }
        }}
        onPointerUp={(e) => (e.target as HTMLElement).releasePointerCapture(e.pointerId)}
        whileTap={{ cursor: "grabbing" }}
      />

      <motion.div
        style={{ display: "none" }}
        animate={{}}
        onUpdate={() => {
          const v = angularVelocity.get();
          if (Math.abs(v) > 0.0001) {
            angle.set(angle.get() + v * 16);
            angularVelocity.set(v * 0.985);
          }
        }}
      />
    </div>
  );
}

}
